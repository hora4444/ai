import os
import json

BASE_DIR = r"C:\ai\source\soloproject\output\questions"

def scan_mock_exams(grade):
    # 예: C:\ai\source\soloproject\output\questions\g1\jsonl
    jsonl_dir = os.path.join(BASE_DIR, grade, "jsonl")
    
    for file_name in os.listdir(jsonl_dir):
        if file_name.endswith(".jsonl"):
            jsonl_path = os.path.join(jsonl_dir, file_name)
            
            with open(jsonl_path, 'r', encoding='utf-8') as f:
                for line in f:
                    data = json.loads(line)
                    # JSONL 내의 assets/path 추출
                    for asset in data.get('assets', []):
                        rel_path = asset.get('path') # "output/questions/assets/..."
                        
                        # 실제 시스템 절대 경로로 변환 (중요!)
                        # 'output/questions/' 부분을 잘라내고 BASE_DIR와 결합하는 로직 필요
                        pure_path = rel_path.replace("output/questions/", "")
                        abs_image_path = os.path.join(BASE_DIR, pure_path)
                        
                        if os.path.exists(abs_image_path):
                            print(f"[OK] {data['id']}: 이미지 확인됨")
                        else:
                            print(f"[ERROR] {data['id']}: 이미지 없음 -> {abs_image_path}")
